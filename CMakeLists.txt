cmake_minimum_required(VERSION 3.20)
project(PCSXReCompressedConverter VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg SDK prefix for desktop builds")
set(LIBCHDR_SOURCE "" CACHE PATH "Optional libchdr checkout for standalone CD CHD import")
set(OPUS_SOURCE "" CACHE PATH "Optional Opus checkout for a fixed-point static build")

find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h HINTS "${FFMPEG_ROOT}/include" REQUIRED)
find_library(AVCODEC_LIBRARY avcodec HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
find_library(AVUTIL_LIBRARY avutil HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY zstd REQUIRED)

add_library(psxrc_reader STATIC src/audio.c src/bs.c src/codec.c src/filters.c src/reader.c src/str.c src/xa.c)
set_target_properties(psxrc_reader PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(psxrc_reader PUBLIC include PRIVATE src "${FFMPEG_INCLUDE_DIR}" "${ZSTD_INCLUDE_DIR}")
target_link_libraries(psxrc_reader PUBLIC "${AVCODEC_LIBRARY}" "${AVUTIL_LIBRARY}" "${ZSTD_LIBRARY}")
if(OPUS_SOURCE)
    function(add_fixed_point_opus)
        set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
        set(BUILD_TESTING OFF)
        set(OPUS_FIXED_POINT ON)
        set(OPUS_BUILD_PROGRAMS OFF)
        set(OPUS_BUILD_TESTING OFF)
        set(OPUS_DISABLE_INTRINSICS ON)
        set(opus_checkout "${OPUS_SOURCE}")
        if(CMAKE_SIZEOF_VOID_P EQUAL 4)
            set(PSXRC_OPUS_SOURCE "${OPUS_SOURCE}")
            set(PSXRC_OPUS_DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/opus-reference")
            include(cmake/prepare-reference-opus.cmake)
            set(opus_checkout "${PSXRC_OPUS_DESTINATION}")
        endif()
        add_subdirectory("${opus_checkout}" opus EXCLUDE_FROM_ALL)
    endfunction()
    add_fixed_point_opus()
    set_target_properties(opus PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(psxrc_reader PUBLIC opus)
else()
    find_path(OPUS_INCLUDE_DIR opus.h PATH_SUFFIXES opus REQUIRED)
    find_library(OPUS_LIBRARY opus REQUIRED)
    target_include_directories(psxrc_reader PRIVATE "${OPUS_INCLUDE_DIR}")
    target_link_libraries(psxrc_reader PUBLIC "${OPUS_LIBRARY}")
endif()

add_executable(psxrc src/main.c src/audio_convert.c src/file_io.c src/source.c src/video_convert.c src/writer.c)
target_include_directories(psxrc PRIVATE "${FFMPEG_INCLUDE_DIR}" "${ZSTD_INCLUDE_DIR}")
target_link_libraries(psxrc PRIVATE psxrc_reader)
if(MINGW)
    target_link_options(psxrc PRIVATE -municode)
endif()
if(LIBCHDR_SOURCE)
    set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
    set(CHDR_WANT_TESTS OFF)
    add_subdirectory("${LIBCHDR_SOURCE}" chdr EXCLUDE_FROM_ALL)
    target_sources(psxrc PRIVATE src/source_chd.c)
    target_compile_definitions(psxrc PRIVATE PSXRC_HAVE_CHD)
    target_link_libraries(psxrc PRIVATE chdr-static)
endif()
if(NOT WIN32)
    target_compile_definitions(psxrc PRIVATE _FILE_OFFSET_BITS=64 _POSIX_C_SOURCE=200809L)
endif()

if(WIN32 AND CMAKE_HOST_WIN32)
    set(PSXRC_RUNTIME_DLL_DIRS "" CACHE STRING "Additional directories containing Windows runtime DLLs")
    set(runtime_dirs ${PSXRC_RUNTIME_DLL_DIRS})
    foreach(library AVCODEC_LIBRARY AVUTIL_LIBRARY ZSTD_LIBRARY OPUS_LIBRARY)
        if(${library})
            get_filename_component(library_dir "${${library}}" DIRECTORY)
            get_filename_component(bin_dir "${library_dir}/../bin" ABSOLUTE)
            list(APPEND runtime_dirs "${library_dir}" "${bin_dir}")
        endif()
    endforeach()
    get_filename_component(compiler_dir "${CMAKE_C_COMPILER}" DIRECTORY)
    list(APPEND runtime_dirs "${compiler_dir}")
    list(REMOVE_DUPLICATES runtime_dirs)
    set(runtime_script "${CMAKE_CURRENT_SOURCE_DIR}/windows/stage-runtime.cmake")
    set(runtime_command "${CMAKE_COMMAND}" "-DPSXRC_EXECUTABLE=$<TARGET_FILE:psxrc>"
        "-DPSXRC_SCAN_DIR=${CMAKE_CURRENT_BINARY_DIR}/CMakeFiles/psxrc-runtime/$<CONFIG>")
    if(MINGW)
        list(APPEND runtime_command "-DPSXRC_OBJDUMP=${CMAKE_OBJDUMP}")
    endif()
    set_property(TARGET psxrc APPEND PROPERTY LINK_DEPENDS "${runtime_script}")
    add_custom_command(TARGET psxrc POST_BUILD
        COMMAND ${runtime_command} "-DPSXRC_SEARCH_DIRS=${runtime_dirs}"
            "-DPSXRC_DESTINATION=$<TARGET_FILE_DIR:psxrc>" -P "${runtime_script}"
        VERBATIM)

    find_program(PSXRC_POWERSHELL NAMES powershell pwsh
        HINTS "$ENV{SystemRoot}/System32/WindowsPowerShell/v1.0")
    if(PSXRC_POWERSHELL)
        set(bundle_dir "${CMAKE_CURRENT_BINARY_DIR}/windows-bundle/$<CONFIG>")
        set(gui_source_dir "${CMAKE_CURRENT_SOURCE_DIR}/windows")
        add_custom_command(OUTPUT "${bundle_dir}/psxrc-gui.exe"
            COMMAND "${CMAKE_COMMAND}" -E make_directory "${bundle_dir}"
            COMMAND "${PSXRC_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
                -File "${gui_source_dir}/build-gui.ps1" -OutputPath "${bundle_dir}/psxrc-gui.exe"
            DEPENDS "${gui_source_dir}/build-gui.ps1" "${gui_source_dir}/ConverterForm.cs"
            VERBATIM)
        add_custom_target(psxrc_windows_bundle
            COMMAND ${runtime_command} "-DPSXRC_SEARCH_DIRS=${runtime_dirs}"
                "-DPSXRC_DESTINATION=${bundle_dir}" -P "${runtime_script}"
            DEPENDS psxrc "${bundle_dir}/psxrc-gui.exe"
            VERBATIM)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(psxrc_make_str tests/make_str.c)
    target_include_directories(psxrc_make_str PRIVATE src)
    target_link_libraries(psxrc_make_str PRIVATE psxrc_reader)
    add_executable(psxrc_make_xa tests/make_xa.c)
    target_include_directories(psxrc_make_xa PRIVATE src)
    target_link_libraries(psxrc_make_xa PRIVATE psxrc_reader)
    if(NOT WIN32)
        target_link_libraries(psxrc_make_xa PRIVATE m)
    endif()
    if(LIBCHDR_SOURCE)
        add_executable(psxrc_bench tests/bench_read.c src/file_io.c src/source.c src/source_chd.c)
        target_include_directories(psxrc_bench PRIVATE src)
        target_compile_definitions(psxrc_bench PRIVATE PSXRC_HAVE_CHD)
        if(NOT WIN32)
            target_compile_definitions(psxrc_bench PRIVATE _FILE_OFFSET_BITS=64 _POSIX_C_SOURCE=200809L)
        endif()
        target_link_libraries(psxrc_bench PRIVATE psxrc_reader chdr-static)
    endif()
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME psxrc_desktop_round_trip COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/round_trip.py"
            "$<TARGET_FILE:psxrc>" "$<TARGET_FILE:psxrc_make_str>" "$<TARGET_FILE:psxrc_make_xa>")
    endif()
endif()

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(psxrc_reader PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(psxrc PRIVATE -Wall -Wextra -Wpedantic)
endif()
